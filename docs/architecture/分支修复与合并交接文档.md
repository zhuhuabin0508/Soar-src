**改动 ① 模块顶部（`logger = logging.getLogger(__name__)` 之后）新增：**

```python
MAX_PROJECTION_ACTIONS_PER_DEVICE = 20

# 设备门面工具治理矩阵（按工具名强制，装配期兜底）。
# main 侧设备模块新增 query_device_actions / execute_device_action 两个 DB 工具行，
# 其预置元数据缺失时默认低危只读——本矩阵按名强制覆盖，不依赖 main 的约定定义。
_DEVICE_FACADE_GOVERNANCE: dict[str, dict[str, Any]] = {
    "query_device_actions": {
        "mutating": False, "risk_level": "low", "danger_level": "low",
        "parallel_safe": True, "deferrable": True, "verification": None,
        "required_permission": ("device", "view"),
    },
    "execute_device_action": {
        "mutating": True, "risk_level": "high", "danger_level": "high",
        "parallel_safe": False, "deferrable": False,
        "verification": "device_action",
        "required_permission": ("device", "edit"),
    },
}


def apply_device_facade_governance(descriptor: ToolDescriptor) -> None:
    """按工具名强制应用设备门面治理矩阵（装配期兜底）。"""
    overrides = _DEVICE_FACADE_GOVERNANCE.get(descriptor.name)
    if overrides is None:
        return
    descriptor.mutating = overrides["mutating"]
    descriptor.risk_level = overrides["risk_level"]
    descriptor.danger_level = overrides["danger_level"]
    descriptor.parallel_safe = overrides["parallel_safe"]
    descriptor.deferrable = overrides["deferrable"]
    descriptor.verification = overrides["verification"]
    descriptor.required_permission = overrides["required_permission"]
```
